class Solution {
public:
    int maxLength(vector<int>& nums) {
        int mx = 2;
        for (auto i = 0; i < nums.size(); i++) {
            int g = nums[i], l = nums[i];
            for (auto j = i + 1; j < nums.size(); j++) {
                g = gcd(g, nums[j]);
                l = lcm(l, nums[j]);
                int l_tmp = l;
                for (int k = i; k <= j; k++) l_tmp /= nums[k];
                if (l_tmp == 1) mx = max(mx, j - i + 1);
                else break;
            }
        }
        return mx;
    }
};
