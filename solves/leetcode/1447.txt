#include <iostream>
#include <vector>
using namespace std;
class Solution {
public:
    int triangularSum(vector<int> nums) {
        int n = nums.size();
        for (int i = 0; i < n - 1; i++) {
            vector<int> tmp(nums.size() - 1);
            for (int j = 1; j < nums.size(); j++) {
                tmp[j - 1] = (nums[j - 1] + nums[j]) % 10;
            }
            nums = tmp;
        }
        return nums[0];
    }
};

signed main() {
    vector<int> w = {9, 2, 0, 4, 5};
    cout << Solution().triangularSum(w) << endl;
}
